    <h3>Job History</h3>

    <button onclick="addJob()">+ Add Job</button>
    <button onclick="calculateScore()">Calculate Score</button>

    <div id="results"></div>

  <script>
    const jobsDiv = document.getElementById("jobs");
    let jobCount = 0;

    function addJob() {
      jobCount++;
      const jobDiv = document.createElement("div");
      jobDiv.classList.add("job-entry");
      jobDiv.innerHTML = `
        <label>Job ${jobCount} Start Date:</label>
        <input type="date" class="start-date" />
        <label>Job ${jobCount} End Date:</label>
        <input type="date" class="end-date" />
      `;
      jobsDiv.appendChild(jobDiv);
    }

    function calculateScore() {
      const gradDateInput = document.getElementById("graduation").value;
      if (!gradDateInput) {
        alert("Please enter your graduation date!");
        return;
      }

      const graduationDate = new Date(gradDateInput);
      const startDates = [...document.querySelectorAll(".start-date")].map(i => new Date(i.value));
      const endDates = [...document.querySelectorAll(".end-date")].map(i => new Date(i.value));

      if (startDates.length === 0) {
        alert("Please add at least one job.");
        return;
      }

      let totalJobDays = 0;
      let numJobs = 0;

      for (let i = 0; i < startDates.length; i++) {
        if (isNaN(startDates[i]) || isNaN(endDates[i])) {
          alert(`Please fill out all job dates (Job ${i + 1}).`);
          return;
        }
        if (endDates[i] <= startDates[i]) {
          alert(`Job ${i + 1}: End date must be after start date.`);
          return;
        }
        totalJobDays += (endDates[i] - startDates[i]) / (1000 * 60 * 60 * 24);
        numJobs++;
      }

      const totalCareerDays = (new Date() - graduationDate) / (1000 * 60 * 60 * 24);
      const avgTenure = totalJobDays / numJobs;
      const hops = numJobs - 1;
      const frequencyFactor = (hops / (totalCareerDays / 365)) * 100;
      const score = Math.round((hops * 1000 / (avgTenure / 30 + 1)) + frequencyFactor * 10);

      // Determine rank and message
      let rank = "";
      let message = "";
      let rankClass = "";

      if (score < 300) {
        rank = "🪨 The Steady Rock";
        message = "You value stability, loyalty, and long-term growth. HR loves you!";
        rankClass = "low";
      } else if (score < 800) {
        rank = "💃 The Smooth Mover";
        message = "You know when to make a smart career move — just enough spice to stay fresh!";
        rankClass = "medium";
      } else if (score < 1500) {
        rank = "🤸 Career Acrobat";
        message = "You leap from opportunity to opportunity with grace and agility!";
        rankClass = "high";
      } else {
        rank = "🚀 The Ultimate Jumping Jack!";
        message = "Your résumé moves faster than a startup pivot — unstoppable motion!";
        rankClass = "extreme";
      }

      const resultsDiv = document.getElementById("results");
      resultsDiv.innerHTML = `
        <h3>📊 Results</h3>
        <p><span class="highlight">Total Career:</span> ${(totalCareerDays / 365).toFixed(1)} years</p>
        <p><span class="highlight">Number of Jobs:</span> ${numJobs}</p>
        <p><span class="highlight">Average Tenure:</span> ${(avgTenure / 365).toFixed(2)} years</p>
        <p><span class="highlight">Job Hops:</span> ${hops}</p>
        <p><span class="highlight">Ultimate Jumping Jack Score:</span> <strong>${score}</strong></p>
        <div class="rank ${rankClass}">
          <p><strong>${rank}</strong></p>
          <p>${message}</p>
        </div>
      `;
    }
  </script>
